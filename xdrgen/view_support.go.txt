// maxDepth is the maximum recursion depth for all view operations:
// sizing, Raw(), array iteration, the Walk, and ValidateFull().
// Matches stellar-core's xdr::marshaling_stack_limit (set in main.cpp).
const maxDepth = 1500

// =============================================================================
// view — the shared representation embedded in every XDR view type.
// =============================================================================

// view is a typed, read-only window into raw XDR bytes:
//
//	d     the value's bytes. When exact is false, d is OPEN-ENDED: it starts
//	      at the value's first byte and extends to the enclosing buffer's end
//	      (the extent is computed lazily by sizing). When exact is true, d is
//	      the value's exact wire extent (already sized), so Raw() is a slice
//	      operation.
//
// Navigation returns sub-views (windows further into the same buffer); no
// intermediate Go structs are built and no bytes are copied. All bounds and
// schema checks happen per access — input is untrusted, and a malformed
// buffer yields a *ViewError, never a panic (Must* methods panic with that
// same *ViewError, recovered by Try/Try0/Try2).
//
// size() computes a value's byte extent. valid() does the same but also
// recursively validates every child (size() may skip fixed-size children for
// performance). Invariants:
//   - valid() ⇒ size(): if valid returns (n, nil), size returns (n, nil).
//   - No panics: every d[off:] is preceded by an off <= len(d) check.
//   - Termination: bounded by maxDepth.
type view struct {
	d     []byte
	exact bool
}

// trimmed trims the view's data to the given size, propagating a sizing
// error. It is the shared tail of every generated Raw() on inexact views.
func (v view) trimmed(sz int, err error) ([]byte, error) {
	if err != nil {
		return nil, err
	}
	if sz > len(v.d) {
		return nil, viewErrShortBuffer(0, fmt.Sprintf("size %d exceeds data length %d", sz, len(v.d)))
	}
	return v.d[:sz], nil
}

// copied returns a detached copy of this view's exact wire bytes.
func (v view) copied(sz int, err error) (view, error) {
	var raw []byte
	if v.exact {
		raw = v.d
	} else {
		raw, err = v.trimmed(sz, err)
		if err != nil {
			return view{}, err
		}
	}
	c := make([]byte, len(raw))
	copy(c, raw)
	return view{d: c, exact: true}, nil
}

// =============================================================================
// Must / Try — the blessed panic-recover pair for chained navigation.
// =============================================================================

// mustView panics with err, which every generated accessor produces as a
// *ViewError — the sentinel Try/Try0/Try2 recover. Callers of Must* methods
// MUST recover via those helpers in the SAME goroutine; a Must panic crossing
// a goroutine boundary is not recoverable by a Try in the parent (Go panics
// do not propagate across goroutines) and will crash the process.
func mustView(err error) {
	panic(err)
}

// Try0 runs fn and converts a Must* panic (the *ViewError sentinel) into an
// ordinary error. Any other panic value is re-panicked untouched. fn must not
// spawn goroutines that call Must*: a panic in another goroutine cannot be
// recovered here and will crash the process.
func Try0(fn func()) (err error) {
	defer func() {
		if r := recover(); r != nil {
			ve, ok := r.(*ViewError)
			if !ok {
				panic(r)
			}
			err = ve
		}
	}()
	fn()
	return nil
}

// Try is Try0 for a single-result fn.
func Try[T any](fn func() T) (v T, err error) {
	err = Try0(func() { v = fn() })
	return v, err
}

// Try2 is Try0 for a two-result fn.
func Try2[A, B any](fn func() (A, B)) (a A, b B, err error) {
	err = Try0(func() { a, b = fn() })
	return a, b, err
}

// =============================================================================
// Error types
// =============================================================================

type ViewError struct {
	Kind   ViewErrorKind
	Offset uint32
	Detail string
}

type ViewErrorKind int

const (
	ViewErrShortBuffer ViewErrorKind = iota
	ViewErrWrongDiscriminant
	ViewErrUnknownDiscriminant
	ViewErrValueOverflow
	ViewErrArrayCountExceedsData
	ViewErrMaxDepth
	ViewErrOpaqueExceedsMax
	ViewErrBadBoolValue
	ViewErrArrayCountExceedsMax
	ViewErrNonZeroPadding
)

func (e *ViewError) Error() string {
	return fmt.Sprintf("xdr view: %s at offset %d: %s", e.Kind.String(), e.Offset, e.Detail)
}

func (k ViewErrorKind) String() string {
	switch k {
	case ViewErrShortBuffer: return "short buffer"
	case ViewErrWrongDiscriminant: return "wrong union discriminant"
	case ViewErrUnknownDiscriminant: return "unknown union discriminant"
	case ViewErrValueOverflow: return "value overflows"
	case ViewErrArrayCountExceedsData: return "array count exceeds remaining data"
	case ViewErrMaxDepth: return "maximum recursion depth exceeded"
	case ViewErrOpaqueExceedsMax: return "opaque/string length exceeds schema max"
	case ViewErrBadBoolValue: return "bool value not 0 or 1"
	case ViewErrArrayCountExceedsMax: return "array count exceeds schema max"
	case ViewErrNonZeroPadding: return "non-zero padding byte"
	default: return "unknown error"
	}
}

func viewErrShortBuffer(off uint32, detail string) error {
	return &ViewError{Kind: ViewErrShortBuffer, Offset: off, Detail: detail}
}
func viewErrWrongDiscriminant(off uint32, got, want int32) error {
	return &ViewError{Kind: ViewErrWrongDiscriminant, Offset: off, Detail: fmt.Sprintf("got %d, want %d", got, want)}
}
func viewErrUnknownDiscriminant(off uint32, disc int32) error {
	return &ViewError{Kind: ViewErrUnknownDiscriminant, Offset: off, Detail: fmt.Sprintf("discriminant %d", disc)}
}
func viewErrValueOverflow(off uint32, what string) error {
	return &ViewError{Kind: ViewErrValueOverflow, Offset: off, Detail: what}
}
func viewErrArrayCountExceedsData(off uint32, count, remaining int) error {
	return &ViewError{Kind: ViewErrArrayCountExceedsData, Offset: off, Detail: fmt.Sprintf("count %d, remaining bytes %d", count, remaining)}
}
func viewErrMaxDepth(off uint32) error {
	return &ViewError{Kind: ViewErrMaxDepth, Offset: off}
}
func viewErrOpaqueExceedsMax(off uint32, length, maxLen uint32) error {
	return &ViewError{Kind: ViewErrOpaqueExceedsMax, Offset: off, Detail: fmt.Sprintf("length %d exceeds max %d", length, maxLen)}
}
func viewErrBadBoolValue(off uint32, val uint32) error {
	return &ViewError{Kind: ViewErrBadBoolValue, Offset: off, Detail: fmt.Sprintf("value %d", val)}
}
func viewErrArrayCountExceedsMax(off uint32, count int, maxCount uint32) error {
	return &ViewError{Kind: ViewErrArrayCountExceedsMax, Offset: off, Detail: fmt.Sprintf("count %d exceeds max %d", count, maxCount)}

}

// errArrayTraverseOOB is a preallocated error used by arrayTraverse to keep
// that function inlineable (error constructors allocate + inflate inline cost).
var errArrayTraverseOOB = &ViewError{Kind: ViewErrShortBuffer, Detail: "array element offset exceeds data"}

const maxInt32 = int(^uint32(0) >> 1)

func safeUint32ToInt(val uint32, off uint32) (int, error) {
	if uint(val) > uint(maxInt32) {
		return 0, viewErrValueOverflow(off, "length or count overflows int32")
	}
	return int(val), nil
}

// =============================================================================
// VarOpaqueView — opaque / string
// Wire layout: [4-byte length][payload][padding to 4-byte boundary]
// =============================================================================

type VarOpaqueView struct{ view }

func NewVarOpaqueView(b []byte) VarOpaqueView { return VarOpaqueView{view{d: b}} }

var zeroPad [3]byte

func (v VarOpaqueView) Value() ([]byte, error) {
	if len(v.d) < 4 { return nil, viewErrShortBuffer(0, "need 4 bytes for length") }
	length, err := safeUint32ToInt(binary.BigEndian.Uint32(v.d[:4]), 0)
	if err != nil { return nil, err }
	// Compute total in int64 to avoid overflow on 32-bit platforms where
	// length may reach maxInt32 and (length + 3) would wrap to negative.
	total64 := 4 + ((int64(length) + 3) &^ 3)
	if total64 > int64(len(v.d)) { return nil, viewErrShortBuffer(0, fmt.Sprintf("need %d padded bytes", total64)) }
	total := int(total64)
	padded := total - 4
	if !bytes.Equal(v.d[4+length:total], zeroPad[:padded-length]) {
		return nil, viewErrNonZeroPadding(uint32(4 + length))
	}
	return v.d[4 : 4+length], nil
}

func sizeVarOpaqueView(d []byte, _ int) (int, error) {
	if len(d) < 4 { return 0, viewErrShortBuffer(0, "need 4 bytes for length") }
	length, err := safeUint32ToInt(binary.BigEndian.Uint32(d[:4]), 0)
	if err != nil { return 0, err }
	total64 := 4 + ((int64(length) + 3) &^ 3)
	if total64 > int64(len(d)) {
		return 0, viewErrShortBuffer(0, fmt.Sprintf("need %d padded bytes, have %d", total64-4, len(d)-4))
	}
	return int(total64), nil
}
// MustValue is Value panicking with the *ViewError sentinel (recover via Try*).
func (v VarOpaqueView) MustValue() []byte {
	val, err := v.Value()
	if err != nil { mustView(err) }
	return val
}
func (v VarOpaqueView) size(depth int) (int, error) { return sizeVarOpaqueView(v.d, depth) }


func validVarOpaqueView(d []byte, depth int) (int, error) { return VarOpaqueView{view{d: d}}.valid(depth) }
func (v VarOpaqueView) valid(_ int) (int, error) {
	if _, err := v.Value(); err != nil { return 0, err }
	return v.size(0)
}

// =============================================================================
// Array helpers — shared runtime for generated array view types.
// =============================================================================

// arrayViewCount reads the 4-byte array count from data.
// If maxCount > 0, it also validates that the count does not exceed the schema bound.
func arrayViewCount(data []byte, maxCount int) (int, error) {
	if len(data) < 4 { return 0, viewErrShortBuffer(0, "need 4 bytes for array count") }
	count, err := safeUint32ToInt(binary.BigEndian.Uint32(data[:4]), 0)
	if err != nil { return 0, err }
	if maxCount > 0 && count > maxCount {
		return 0, viewErrArrayCountExceedsMax(0, count, uint32(maxCount))
	}
	return count, nil
}

// arrayViewCountChecked reads the 4-byte array count from data and validates it
// against both the schema maximum (maxCount, if > 0) and the remaining buffer
// (the OOM guard: an array whose declared count cannot fit the data is rejected
// before anything allocates from that count).
//
// The buffer check rejects any count whose elements cannot fit the data, in
// O(1). It is written as a division to be overflow-safe: a wire count can be up
// to ~4.3e9, so count*minElemW could overflow; (len(data)-4)/minElemW cannot,
// since minElemW >= 1 and both operands are non-negative.
func arrayViewCountChecked(data []byte, maxCount, minElemW int) (int, error) {
	count, err := arrayViewCount(data, maxCount)
	if err != nil { return 0, err }
	if count > (len(data)-4)/minElemW {
		return 0, viewErrArrayCountExceedsData(0, count, len(data)-4)
	}
	return count, nil
}

// arrayTraverse iterates count elements starting at startOff, calling fn on each
// element's bytes. Returns the total byte extent. Used by generated size() and
// valid() methods via closures — the Go compiler inlines arrayTraverse, the
// closure, and the inner method call, producing identical machine code to
// hand-written loops.
//
// IMPORTANT: This function MUST remain inlineable (cost < 80 as reported by
// go build -gcflags='-m=2'). If it exceeds the budget, all array size/valid
// operations regress by ~17% due to indirect closure dispatch. Keep it minimal:
// no error constructors (use errArrayTraverseOOB), no string literals, no
// bounds checks with messages.
//
// The uint(off) > uint(len(data)) check catches both positive out-of-bounds
// (a malformed size() returning more than available) and wraparound into
// negative on 32-bit platforms (where int is 32-bit), turning either into a
// clean *ViewError return rather than a runtime slice panic.
func arrayTraverse(data []byte, count, startOff int, fn func([]byte) (int, error)) (int, error) {
	off := startOff
	for k := 0; k < count; k++ {
		if uint(off) > uint(len(data)) {
			return 0, errArrayTraverseOOB
		}
		sz, err := fn(data[off:])
		if err != nil { return 0, err }
		off += sz
	}
	return off, nil
}

// =============================================================================
// Walk support (tier 2)
// =============================================================================

// stopWalk is the type behind ErrStopWalk.
type stopWalk struct{}

func (stopWalk) Error() string { return "xdr: walk stopped by subscriber" }

// ErrStopWalk, returned from any Walk callback, stops the walk cleanly: the
// WalkX call returns nil. Any other non-nil callback error aborts the walk
// and is returned verbatim.
var ErrStopWalk error = stopWalk{}
